## Key excerpt

```go
type mockUserRepo struct {
	users map[int]*User
	err   error
}

func (m *mockUserRepo) FindById(id int) (*User, error) {
	if m.err != nil {
		return nil, m.err
	}
	u, ok := m.users[id]
	if !ok {
		return nil, errors.New("user not found")
	}
	return u, nil
}
```
